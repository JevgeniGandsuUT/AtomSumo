 221:     var holdButtons = [

 227:     var stopButton = document.getElementById("stop");
 228:     var activeDirection = "stop";
 229: 
 230:     function send(path) {
 231:       fetch(path, { cache: "no-store" }).catch(function () {});
 232:     }
 233: 
 234:     function startMove(dir, button) {
 235:       if (activeDirection === dir) return;
 236:       activeDirection = dir;
 237:       button.classList.add("active");
 238:       send("/move?dir=" + dir);
 239:     }
 240: 
 241:     function clearPadButtons() {
 242:       var padButtons = document.querySelectorAll(".pad button");
 243:       for (var i = 0; i < padButtons.length; i++) {
 244:         padButtons[i].classList.remove("active");
 245:       }
 246:     }
 247: 
 248:     function stopMove(button) {
 249:       if (button) {
 250:         button.classList.remove("active");
 251:       }
 252:       if (activeDirection === "stop") return;
 253:       activeDirection = "stop";
 254:       clearPadButtons();
 255:       send("/move?dir=stop");
 256:     }
 257: 
 258:     function bindHold(button, dir) {
 259:       button.addEventListener("pointerdown", function (e) {
 260:         e.preventDefault();
 261:         startMove(dir, button);
 262:       });
 263: 
 264:       var releaseEvents = ["pointerup", "pointercancel", "pointerleave"];
 265:       for (var i = 0; i < releaseEvents.length; i++) {
 266:         (function (eventName) {
 267:           button.addEventListener(eventName, function (e) {
 268:             e.preventDefault();
 269:             stopMove(button);
 270:           });
 271:         })(releaseEvents[i]);
 272:       }
 273:     }
 274: 
 275:     for (var j = 0; j < holdButtons.length; j++) {
 276:       var button = document.getElementById(holdButtons[j].id);
 277:       bindHold(button, holdButtons[j].dir);
